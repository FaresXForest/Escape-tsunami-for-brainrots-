local function setClipboard(text)
    if (syn and syn.write_clipboard) then syn.write_clipboard(text)
    elseif (setclipboard) then setclipboard(text)
    else
        local UserInputService = game:GetService("UserInputService")
        local old = UserInputService.TextBoxFocused
        local box = Instance.new("TextBox", MainGui)
        box.Text = text
        box:CaptureFocus()
        UserInputService.TextBoxFocused = old
        box:Destroy()
    end
end

    CopyButton.BackgroundColor3 = Color3.fromRGB(60, 180, 110)
    task.wait(2)
    CopyButton.Text = originalText
    CopyButton.BackgroundColor3 = originalColor
end)

end
